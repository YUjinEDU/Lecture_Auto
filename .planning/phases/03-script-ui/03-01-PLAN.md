class FontInfo(BaseModel):
    name: str | None = None
    size_pt: float | None = None
    bold: bool | None = None
    italic: bool | None = None

class TextRun(BaseModel):
    text: str
    font: FontInfo

class TextParagraph(BaseModel):
    runs: list[TextRun]
    full_text: str

class ShapeRecord(BaseModel):
    shape_id: int
    name: str
    z_order: int
    left: int    # EMU
    top: int
    width: int
    height: int
    content_source: Literal["text", "image", "chart", "smartart", "ole_object", "table", "other"]
    has_image: bool = False
    has_text: bool = False
    text_role: Literal["title", "body", "other"] | None = None
    paragraphs: list[TextParagraph] = Field(default_factory=list)

class SlideRecord(BaseModel):
    slide_index: int
    slide_number: int
    png_path: str
    shapes: list[ShapeRecord]

    2. Create `lecture_auto/pipeline/parser_pdf.py` with these functions:

       `is_title_span(span: dict, page_height: float) -> bool`:
       Per D-08, return True if `span["bbox"][1] < page_height * 0.25` AND (`span["size"] >= 18.0` OR `bool(span["flags"] & 16)`).
       Also check if font name contains "Bold" as secondary signal (per RESEARCH.md Pitfall 1).

       `parse_pdf_page(page, page_num: int) -> SlideRecord`:
       Call `page.get_text("dict")`. Iterate blocks:
       - type==0 (text): Build TextParagraph/TextRun from spans. FontInfo from span["font"], round(span["size"], 1), flags&16 for bold, flags&2 for italic.
         Determine is_title by checking if ANY span in the block passes is_title_span.
         Convert bbox from points to EMU: `int(value * 12700)`.
         Create ShapeRecord with content_source="text", has_text=True, text_role="title"|"body".
       - type==1 (image): Create ShapeRecord with content_source="image", has_image=True, bbox converted to EMU.
       Return SlideRecord(slide_index=page_num, slide_number=page_num+1, png_path=f"slide_{page_num+1:03d}.png", shapes=shapes).

       `parse_pdf(pdf_path: str | Path) -> list[SlideRecord]`:
       Open with `pymupdf.open(str(pdf_path))`. Iterate pages, call parse_pdf_page for each. Close doc. Return list.

       `extract_tables_fallback(pdf_path: str | Path, page_num: int) -> list[list[list[str]]]`:
       Per D-09, use `pdfplumber.open(str(pdf_path))`, access `pdf.pages[page_num].extract_tables()`. Return tables or empty list.

       `parse_input(input_path: Path) -> list[SlideRecord]`:
       Per D-10, if suffix is `.pdf`, call parse_pdf directly. If suffix is `.pptx`, convert via LibreOffice subprocess (`soffice --headless --convert-to pdf`) to a temp dir, then parse_pdf on the output. Raise ValueError for unsupported extensions.

       - Use `pymupdf.open()` (empty doc) + `doc.new_page()` + `page.insert_text()` to create test PDFs.
       - Test title detection, body classification, EMU conversion, image blocks, table fallback.
