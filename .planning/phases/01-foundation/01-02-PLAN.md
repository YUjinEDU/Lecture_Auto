```python
from pptx import Presentation
from pptx.enum.shapes import MSO_SHAPE_TYPE, PP_PLACEHOLDER
from pptx.util import Pt
from pathlib import Path
from io import BytesIO
from lecture_auto.schemas.manifest import (
    ShapeRecord, SlideRecord, FontInfo, TextRun, TextParagraph,
)

def extract_font_info(run) -> FontInfo:
    """Extract font metadata from a python-pptx Run object."""
    font = run.font
    size_pt = None
    if font.size is not None:
        size_pt = Pt(font.size).pt if hasattr(Pt(font.size), 'pt') else font.size / 12700  # EMU to pt
    return FontInfo(
        name=font.name,
        size_pt=round(font.size / 12700, 1) if font.size else None,
        bold=font.bold,
        italic=font.italic,
    )

def extract_text_content(shape) -> dict:
    """Extract structured text from a shape with text_frame."""
    paragraphs = []
    for para in shape.text_frame.paragraphs:
        runs = []
        for run in para.runs:
            runs.append(TextRun(text=run.text, font=extract_font_info(run)))
        full_text = "".join(r.text for r in runs)
        if runs:  # skip empty paragraphs with no runs
            paragraphs.append(TextParagraph(runs=runs, full_text=full_text))

    # Determine text_role from placeholder type
    text_role = "other"
    if shape.is_placeholder:
        ph_type = shape.placeholder_format.type
        if ph_type in (PP_PLACEHOLDER.TITLE, PP_PLACEHOLDER.CENTER_TITLE, PP_PLACEHOLDER.SUBTITLE):
            text_role = "title"
        elif ph_type in (PP_PLACEHOLDER.BODY, PP_PLACEHOLDER.OBJECT):
            text_role = "body"

    return {
        "has_text": True,
        "text_role": text_role,
        "paragraphs": paragraphs,
    }

def classify_shape(shape, z_order: int) -> ShapeRecord:
    """Classify a python-pptx shape into a ShapeRecord."""
    base = {
        "shape_id": shape.shape_id,
        "name": shape.name,
        "z_order": z_order,
        "left": shape.left or 0,
        "top": shape.top or 0,
        "width": shape.width or 0,
        "height": shape.height or 0,
    }

    # SmartArt detection (python-pptx 1.0+ has has_smart_art)
    if hasattr(shape, "has_smart_art") and shape.has_smart_art:
        return ShapeRecord(**base, content_source="smartart")
    if shape.shape_type == MSO_SHAPE_TYPE.IGX_GRAPHIC:
        return ShapeRecord(**base, content_source="smartart")

    # Chart detection
    if hasattr(shape, "has_chart") and shape.has_chart:
        return ShapeRecord(**base, content_source="chart")

    # OLE object detection
    if shape.shape_type in (
        MSO_SHAPE_TYPE.EMBEDDED_OLE_OBJECT,
        MSO_SHAPE_TYPE.LINKED_OLE_OBJECT,
    ):
        return ShapeRecord(**base, content_source="ole_object")

    # Table detection
    if hasattr(shape, "has_table") and shape.has_table:
        return ShapeRecord(**base, content_source="table")

    # Picture detection
    if shape.shape_type == MSO_SHAPE_TYPE.PICTURE:
        return ShapeRecord(**base, content_source="image", has_image=True)

    # Text detection
    if shape.has_text_frame:
        text_data = extract_text_content(shape)
        return ShapeRecord(**base, content_source="text", **text_data)

    return ShapeRecord(**base, content_source="other")

def parse_pptx(pptx_data: bytes | Path) -> list[SlideRecord]:
    """Parse PPTX file into a list of SlideRecord objects.

    Args:
        pptx_data: Either raw bytes of a PPTX file or a Path to one.

    Returns:
        List of SlideRecord, one per slide, 0-indexed.
    """
    if isinstance(pptx_data, Path):
        prs = Presentation(str(pptx_data))
    else:
        prs = Presentation(BytesIO(pptx_data))

    slides = []
    for slide_idx, slide in enumerate(prs.slides):
        shapes = []
        for z_order, shape in enumerate(slide.shapes):
            shapes.append(classify_shape(shape, z_order))

        slides.append(SlideRecord(
            slide_index=slide_idx,
            slide_number=slide_idx + 1,
            png_path=f"slide_{slide_idx + 1:03d}.png",
            shapes=shapes,
        ))

    return slides
```

Note on font.size: python-pptx stores font.size in EMU (English Metric Units). 1 pt = 12700 EMU. Convert with `font.size / 12700` when font.size is not None. Do NOT use `Pt(font.size).pt` — `Pt()` is a constructor that takes points, not EMU.
