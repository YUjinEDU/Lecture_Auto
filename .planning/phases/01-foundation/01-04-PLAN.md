```python
"""FastAPI dependencies for authentication and authorization."""
import os
import jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials

security = HTTPBearer()

def get_current_professor(
    cred: HTTPAuthorizationCredentials = Depends(security),
) -> dict:
    """Validate Supabase JWT and enforce professor role.

    Supabase JWTs use HS256 with audience="authenticated".
    Professor role is stored in app_metadata.role (NOT user_metadata, which is user-editable).

    Raises:
        HTTPException 401: Invalid/expired token or wrong audience
        HTTPException 403: Valid token but role is not "professor"
    """
    try:
        payload = jwt.decode(
            cred.credentials,
            os.environ["SUPABASE_JWT_SECRET"],
            algorithms=["HS256"],
            audience="authenticated",
        )
    except jwt.ExpiredSignatureError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Token expired",
        )
    except jwt.PyJWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token",
        )

    role = payload.get("app_metadata", {}).get("role")
    if role != "professor":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Professor role required",
        )

    return payload
```

```python
"""PPTX upload endpoint - synchronous processing (Phase 1, per D-09).

Accepts PPTX file + form metadata in a single multipart request (per D-08).
Runs parsing and rendering inline, returns manifest JSON.
Duplicate detection via SHA-256 hash (per D-10).
"""
import hashlib
import logging
import uuid
from datetime import datetime, timezone
from pathlib import Path

from fastapi import APIRouter, UploadFile, File, Form, Depends, HTTPException, status

from lecture_auto.api.deps import get_current_professor
from lecture_auto.schemas.manifest import SlideManifest, LectureStyle
from lecture_auto.schemas.request import UploadResponse
from lecture_auto.storage.jobs import JobPaths
from lecture_auto.pipeline.parser import parse_pptx
from lecture_auto.pipeline.renderer import render_slides

router = APIRouter()

# In-memory duplicate tracking for Phase 1 (Supabase jobs table in Phase 2)
_file_hashes: dict[str, str] = {}  # sha256 -> job_id

def _setup_job_logger(logs_dir: Path, job_id: str) -> logging.Logger:
    """Create per-job file logger (per INFRA-05)."""
    job_logger = logging.getLogger(f"job.{job_id}")
    job_logger.setLevel(logging.INFO)
    handler = logging.FileHandler(logs_dir / f"{job_id}.log")
    handler.setFormatter(logging.Formatter(
        "%(asctime)s %(levelname)s %(message)s"
    ))
    job_logger.addHandler(handler)
    return job_logger

@router.post("/upload", response_model=UploadResponse)
async def upload_pptx(
    file: UploadFile = File(..., description="PPTX file to process"),
    lecture_name: str = Form(..., description="Lecture title"),
    subject_name: str = Form(..., description="Subject/course name"),
    target_audience: str = Form(..., description="Target audience description"),
    target_minutes: int = Form(..., ge=1, le=300, description="Target lecture duration in minutes"),
    style_density: str = Form(..., description="concise or detailed"),
    style_tone: str = Form(..., description="formal or casual"),
    style_approach: str = Form(..., description="explanatory or socratic"),
    style_supplement: str = Form("", description="Free-text style supplement"),
    professor: dict = Depends(get_current_professor),
):
    """Upload PPTX and run synchronous parse + render pipeline.

    Per D-09: Phase 1 is synchronous. Returns manifest in same response.
    Per D-08: File + metadata in single multipart request.
    Per D-10: Duplicate SHA-256 returns 409 with existing job_id.
    """
    # Validate file extension
    if not file.filename or not file.filename.lower().endswith(".pptx"):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="File must be a .pptx file",
        )

    # Validate style enum values
    if style_density not in ("concise", "detailed"):
        raise HTTPException(status_code=422, detail="style_density must be 'concise' or 'detailed'")
    if style_tone not in ("formal", "casual"):
        raise HTTPException(status_code=422, detail="style_tone must be 'formal' or 'casual'")
    if style_approach not in ("explanatory", "socratic"):
        raise HTTPException(status_code=422, detail="style_approach must be 'explanatory' or 'socratic'")

    # Read file and compute SHA-256
    pptx_bytes = await file.read()
    file_sha256 = hashlib.sha256(pptx_bytes).hexdigest()

    # Duplicate detection (per D-10)
    if file_sha256 in _file_hashes:
        existing_job_id = _file_hashes[file_sha256]
        return UploadResponse(
            job_id=existing_job_id,
            status="duplicate_warning",
            slide_count=0,
            message=f"같은 파일이 있습니다. 기존 job: {existing_job_id}",
            manifest_path="",
            duplicate_of=existing_job_id,
        )

    # Create job
    job_id = str(uuid.uuid4())
    paths = JobPaths(job_id)
    paths.ensure_dirs()

    # Setup per-job logger (per INFRA-05)
    job_logger = _setup_job_logger(paths.logs_dir, job_id)
    job_logger.info("Job started: %s", job_id)
    job_logger.info("File: %s (SHA-256: %s)", file.filename, file_sha256)

    try:
        # Save uploaded file
        input_path = paths.input_dir / file.filename
        input_path.write_bytes(pptx_bytes)
        job_logger.info("Saved PPTX to %s", input_path)

        # Parse PPTX (per PARSE-01, PARSE-02, PARSE-03)
        job_logger.info("Starting PPTX parsing")
        slide_records = parse_pptx(pptx_bytes)
        job_logger.info("Parsed %d slides", len(slide_records))

        # Render slides (per RENDER-01, RENDER-02, RENDER-03)
        job_logger.info("Starting slide rendering")
        png_paths, font_warnings = render_slides(input_path, paths.rendered_dir, job_id)
        job_logger.info("Rendered %d PNGs, %d font warnings", len(png_paths), len(font_warnings))

        # Build manifest
        style = LectureStyle(
            density=style_density,
            tone=style_tone,
            approach=style_approach,
            supplement=style_supplement,
        )
        manifest = SlideManifest(
            job_id=job_id,
            file_sha256=file_sha256,
            lecture_name=lecture_name,
            subject_name=subject_name,
            target_audience=target_audience,
            target_minutes=target_minutes,
            style=style,
            slide_count=len(slide_records),
            slides=slide_records,
            font_warnings=font_warnings,
        )

        # Save manifest JSON (per PARSE-03)
        manifest_json = manifest.model_dump_json(indent=2)
        paths.manifest_path.write_text(manifest_json, encoding="utf-8")
        job_logger.info("Manifest saved to %s", paths.manifest_path)

        # Register hash for duplicate detection
        _file_hashes[file_sha256] = job_id

        job_logger.info("Job completed successfully")

        return UploadResponse(
            job_id=job_id,
            status="completed",
            slide_count=len(slide_records),
            message=f"Successfully parsed and rendered {len(slide_records)} slides",
            manifest_path=str(paths.manifest_path),
        )

    except Exception as e:
        job_logger.error("Job failed: %s", str(e), exc_info=True)
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Pipeline failed: {str(e)}",
        )
    finally:
        # Clean up logger handlers
        for handler in job_logger.handlers[:]:
            handler.close()
            job_logger.removeHandler(handler)
```

```python
"""FastAPI application entry point."""
from contextlib import asynccontextmanager
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from dotenv import load_dotenv
import logging

load_dotenv()

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(name)s %(levelname)s %(message)s",
)

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Application lifespan: startup and shutdown."""
    logging.getLogger(__name__).info("Lecture Auto API starting")
    yield
    logging.getLogger(__name__).info("Lecture Auto API shutting down")

app = FastAPI(
    title="Lecture Auto API",
    description="PPTX to lecture script and audio automation pipeline",
    version="0.1.0",
    lifespan=lifespan,
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],  # Tighten in production
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

from lecture_auto.api.routes import health, upload
app.include_router(health.router, tags=["health"])
app.include_router(upload.router, tags=["upload"])
```
